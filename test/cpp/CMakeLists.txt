cmake_minimum_required(VERSION 3.16)
project(cote_console_tests LANGUAGES CXX)
enable_testing()
add_executable(console_style_tests styles.test.cpp "../../packages/c++/main.cpp")
target_include_directories(console_style_tests PRIVATE mock)
target_compile_definitions(console_style_tests PRIVATE CO_TE_NO_MAIN)
target_compile_features(console_style_tests PRIVATE cxx_std_17)
foreach(case bold underline underline-enabled blink write-failure info-failure get-mode-failure set-mode-failure attribute-failure)
  add_test(NAME ${case} COMMAND console_style_tests ${case})
endforeach()
# Compile the unmocked example against the actual Windows SDK in Windows CI.
if(WIN32)
  add_executable(console_example "../../packages/c++/main.cpp")
  target_compile_features(console_example PRIVATE cxx_std_17)
endif()
if(WIN32)
  add_executable(console_native_tests native.test.cpp "../../packages/c++/main.cpp")
  target_compile_definitions(console_native_tests PRIVATE CO_TE_NO_MAIN)
  target_compile_features(console_native_tests PRIVATE cxx_std_17)
  add_test(NAME native-console-styles COMMAND console_native_tests)
endif()

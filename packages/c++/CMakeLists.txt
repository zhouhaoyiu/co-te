cmake_minimum_required(VERSION 3.16)
project(cote_console LANGUAGES C CXX)
add_library(cote_utf8proc STATIC vendor/utf8proc/utf8proc.c)
target_include_directories(cote_utf8proc PUBLIC vendor/utf8proc)
target_compile_definitions(cote_utf8proc PUBLIC UTF8PROC_STATIC)
if(WIN32)
  add_library(cote_console STATIC main.cpp)
  target_compile_definitions(cote_console PRIVATE CO_TE_NO_MAIN)
  target_compile_features(cote_console PUBLIC cxx_std_17)
  target_include_directories(cote_console PUBLIC .)
  target_link_libraries(cote_console PUBLIC cote_utf8proc)
  add_executable(console_example main.cpp)
  target_compile_features(console_example PRIVATE cxx_std_17)
  target_link_libraries(console_example PRIVATE cote_utf8proc)
endif()
